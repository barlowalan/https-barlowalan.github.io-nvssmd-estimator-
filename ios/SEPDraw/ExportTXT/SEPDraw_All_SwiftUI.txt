struct CableRoute: Codable, Hashable, Identifiable {
    var id = UUID()
    var name: String
    var type: CableType
    var points: [WorldPoint]
    var layerID: UUID
    var slackPercent: Double = 10
    var sourceDeviceID: UUID?
    var destinationDeviceID: UUID?

    func length(scaleFeetPerPoint: Double) -> Double {
        guard points.count > 1 else { return 0 }
        let raw = zip(points, points.dropFirst()).reduce(0.0) { total, pair in
            total + hypot(pair.1.x - pair.0.x, pair.1.y - pair.0.y)
        }
        return raw * scaleFeetPerPoint * (1 + slackPercent / 100)
    }
}

// MARK: - SEPDraw/SEPDrawApp/Services/BOMEngine.swift
import Foundation

enum BOMEngine {
    static func build(project: ProjectDocument) -> [BOMLine] {
        var lines: [String: BOMLine] = [:]
        for device in project.devices where !device.existing {
            let key = device.product.partNumber
            var line = lines[key] ?? BOMLine(id: key, partNumber: key, description: device.product.description, quantity: 0, material: 0, laborHours: 0)
            line.quantity += 1
            line.material += device.product.unitCost
            line.laborHours += device.product.laborHours
            lines[key] = line
        }
        for route in project.cableRoutes {
            let feet = Int(ceil(route.length(scaleFeetPerPoint: project.scaleFeetPerPoint) - 0.000_001))
            let key = "CABLE-\(route.type.rawValue)"
            var line = lines[key] ?? BOMLine(id: key, partNumber: key, description: "\(route.type.rawValue) cable", quantity: 0, material: 0, laborHours: 0)
            line.quantity += feet
            line.material += Decimal(feet) * route.type.costPerFoot
            line.laborHours += Double(feet) / 100 * 1.5
            lines[key] = line
        }
        return lines.values.sorted { $0.partNumber < $1.partNumber }
    }
}

import Foundation

struct DXFCodec {
    enum CodecError: LocalizedError {
        case unsupportedEncoding
        var errorDescription: String? { "The DXF must be an ASCII DXF file. Binary DXF requires a licensed CAD SDK." }
    }

    func decode(_ data: Data) throws -> [VectorPrimitive] {
        guard let text = String(data: data, encoding: .utf8) ?? String(data: data, encoding: .ascii) else { throw CodecError.unsupportedEncoding }
        let raw = text.components(separatedBy: .newlines)
        var pairs: [(Int, String)] = []
        var index = 0
        while index + 1 < raw.count {
            if let code = Int(raw[index].trimmingCharacters(in: .whitespaces)) {
                pairs.append((code, raw[index + 1].trimmingCharacters(in: .whitespaces)))
            }
            index += 2
        }
        var output: [VectorPrimitive] = []
        var i = 0
        while i < pairs.count {
            guard pairs[i].0 == 0 else { i += 1; continue }
            let entity = pairs[i].1.uppercased(); i += 1
            var fields: [Int: [String]] = [:]
            while i < pairs.count && pairs[i].0 != 0 {
                fields[pairs[i].0, default: []].append(pairs[i].1); i += 1
            }
            func d(_ code: Int, _ occurrence: Int = 0) -> Double {
                guard let values = fields[code], occurrence < values.count else { return 0 }
                return Double(values[occurrence]) ?? 0
            }
            switch entity {
            case "LINE": output.append(.line(id: UUID(), start: .init(x: d(10), y: -d(20)), end: .init(x: d(11), y: -d(21))))
            case "CIRCLE": output.append(.circle(id: UUID(), center: .init(x: d(10), y: -d(20)), radius: d(40)))
            case "LWPOLYLINE":
                let xs = fields[10] ?? [], ys = fields[20] ?? []
                let points = zip(xs, ys).map { WorldPoint(x: Double($0.0) ?? 0, y: -(Double($0.1) ?? 0)) }
                let flags = Int(fields[70]?.first ?? "0") ?? 0
                output.append(.polyline(id: UUID(), points: points, closed: (flags & 1) == 1))
            case "TEXT", "MTEXT": output.append(.text(id: UUID(), origin: .init(x: d(10), y: -d(20)), value: fields[1]?.first ?? "", height: d(40)))
            default: break
            }
        }
        return output
    }

    func encode(project: ProjectDocument) -> Data {
        var s = "0\nSECTION\n2\nHEADER\n0\nENDSEC\n0\nSECTION\n2\nENTITIES\n"
        for primitive in project.importedVectors {
            switch primitive {
            case let .line(_, a, b): s += "0\nLINE\n8\nBACKGROUND\n10\n\(a.x)\n20\n\(-a.y)\n11\n\(b.x)\n21\n\(-b.y)\n"
            case let .circle(_, c, r): s += "0\nCIRCLE\n8\nBACKGROUND\n10\n\(c.x)\n20\n\(-c.y)\n40\n\(r)\n"
            case let .polyline(_, points, closed):
                s += "0\nLWPOLYLINE\n8\nBACKGROUND\n90\n\(points.count)\n70\n\(closed ? 1 : 0)\n"
                for p in points { s += "10\n\(p.x)\n20\n\(-p.y)\n" }
            case let .text(_, p, value, height): s += "0\nTEXT\n8\nBACKGROUND\n10\n\(p.x)\n20\n\(-p.y)\n40\n\(height)\n1\n\(value)\n"
            }
        }
        for device in project.devices {
            s += "0\nCIRCLE\n8\nSECURITY-\(device.product.kind.discipline.rawValue)\n10\n\(device.position.x)\n20\n\(-device.position.y)\n40\n12\n"
            s += "0\nTEXT\n8\nSECURITY-LABELS\n10\n\(device.position.x + 14)\n20\n\(-device.position.y)\n40\n8\n1\n\(device.label)\n"
        }
        s += "0\nENDSEC\n0\nEOF\n"
        return Data(s.utf8)
    }
}

struct EngineeringSummary {
    var poeWatts: Double
    var cameraCount: Int
    var storageTB30Days: Double
    var totalCableFeet: Double
    var longestRouteFeet: Double
    var warnings: [String]
}

enum EngineeringEngine {
    static func summarize(_ project: ProjectDocument) -> EngineeringSummary {
        let cameras = project.devices.filter { $0.product.kind.discipline == .video }
        let poe = project.devices.reduce(0) { $0 + $1.product.watts }
        let lengths = project.cableRoutes.map { $0.length(scaleFeetPerPoint: project.scaleFeetPerPoint) }
        var warnings: [String] = []
        for route in project.cableRoutes {
            let feet = route.length(scaleFeetPerPoint: project.scaleFeetPerPoint)
            if [.cat6, .cat6a].contains(route.type), feet > 328 { warnings.append("\(route.name) exceeds the 328 ft copper Ethernet channel limit.") }
        }
        if poe > 370 { warnings.append("Connected device load exceeds a typical 370 W 24-port PoE switch budget.") }
        let storage = Double(cameras.count) * storageTB(mbps: 4, days: 30, dutyCycle: 1)
        return .init(poeWatts: poe, cameraCount: cameras.count, storageTB30Days: storage, totalCableFeet: lengths.reduce(0, +), longestRouteFeet: lengths.max() ?? 0, warnings: warnings)
    }

    static func storageTB(mbps: Double, days: Int, dutyCycle: Double) -> Double {
        mbps * 1_000_000 / 8 * 86_400 * Double(days) * dutyCycle / 1_000_000_000_000
    }

    static func pixelDensity(horizontalPixels: Int = 3840, rangeFeet: Double, fieldOfViewDegrees: Double) -> Double {
        let width = 2 * rangeFeet * tan(fieldOfViewDegrees * .pi / 360)
        return width > 0 ? Double(horizontalPixels) / width : 0
    }

    static func doriLabel(ppf: Double) -> String {
        switch ppf {
        case 250...: "Identify"
        case 125..<250: "Recognize"
        case 62..<125: "Observe"
        case 25..<62: "Detect"
        default: "Monitor"
        }
    }
}

extension UTType {

    static let dwg = UTType(filenameExtension: "dwg") ?? .data

    static let visioSVG = UTType(filenameExtension: "svg") ?? .xml
}

struct FileExchangeService {
    enum ExchangeError: LocalizedError {
        case unsupported(String)
        case invalidProject
        var errorDescription: String? {
            switch self {
            case .unsupported(let detail): detail
            case .invalidProject: "This SEP Draw project is damaged or uses a newer schema."
            }
        }
    }

    func importFile(url: URL, into current: ProjectDocument) async throws -> ProjectDocument {
        let secured = url.startAccessingSecurityScopedResource()
        defer { if secured { url.stopAccessingSecurityScopedResource() } }
        let data = try Data(contentsOf: url)
        let ext = url.pathExtension.lowercased()
        switch ext {
        case "sepdraw":
            let decoder = JSONDecoder(); decoder.dateDecodingStrategy = .iso8601
            let decoded = try decoder.decode(ProjectDocument.self, from: data)
            guard decoded.schemaVersion <= ProjectDocument.schemaVersion else { throw ExchangeError.invalidProject }
            return decoded
        case "pdf":
            guard PDFDocument(data: data) != nil else { throw ExchangeError.unsupported("The selected PDF could not be opened.") }
            var copy = current
            copy.backgroundPDFData = data
            copy.sourceFileData = data
            copy.sourceFileName = url.lastPathComponent
            copy.sourceFileType = "PDF"
            return copy
        case "dxf":
            var copy = current
            copy.importedVectors = try DXFCodec().decode(data)
            copy.sourceFileData = data
            copy.sourceFileName = url.lastPathComponent
            copy.sourceFileType = "DXF"
            return copy
        case "dwg", "vsd", "vsdx":
            // The original is retained losslessly for round-trip export. A production
            // ODA/Visio adapter can replace this reference-only path without touching UI.
            var copy = current
            copy.sourceFileData = data
            copy.sourceFileName = url.lastPathComponent
            copy.sourceFileType = ext.uppercased()
            return copy
        default:
            throw ExchangeError.unsupported("Supported imports are SEP Draw, PDF, ASCII DXF, DWG and VSDX.")
        }
    }

    func projectData(_ project: ProjectDocument) throws -> Data {
        let encoder = JSONEncoder(); encoder.outputFormatting = [.prettyPrinted, .sortedKeys]; encoder.dateEncodingStrategy = .iso8601
        return try encoder.encode(project)
    }

    func csvBOM(_ project: ProjectDocument) -> Data {
        var lines = ["Part Number,Description,Quantity,Material Cost,Labor Hours"]
        for item in BOMEngine.build(project: project) {
            let description = item.description.replacingOccurrences(of: "\"", with: "\"\"")
            lines.append("\(item.partNumber),\"\(description)\",\(item.quantity),\(item.material),\(String(format: "%.2f", item.laborHours))")
        }
        return Data(lines.joined(separator: "\n").utf8)
    }
}

// MARK: - SEPDraw/SEPDrawApp/Services/PDFDrawingRenderer.swift
import UIKit
import PDFKit

enum PDFDrawingRenderer {
    static func render(project: ProjectDocument, pageSize: CGSize = CGSize(width: 1_584, height: 1_224)) -> Data {
        let format = UIGraphicsPDFRendererFormat()
        format.documentInfo = [kCGPDFContextTitle as String: project.name, kCGPDFContextCreator as String: "SEP Draw"]
        let renderer = UIGraphicsPDFRenderer(bounds: CGRect(origin: .zero, size: pageSize), format: format)
        return renderer.pdfData { context in
            context.beginPage()
            let cg = context.cgContext
            UIColor.white.setFill(); cg.fill(CGRect(origin: .zero, size: pageSize))
            if let data = project.backgroundPDFData, let page = PDFDocument(data: data)?.page(at: 0), let ref = page.pageRef {
                let box = page.bounds(for: .mediaBox)
                cg.saveGState(); cg.translateBy(x: 0, y: pageSize.height); cg.scaleBy(x: pageSize.width / box.width, y: -pageSize.height / box.height); cg.drawPDFPage(ref); cg.restoreGState()
            }
            cg.setLineWidth(1); cg.setStrokeColor(UIColor.systemGray.cgColor)
            for vector in project.importedVectors { draw(vector, in: cg) }
            for route in project.cableRoutes {
                guard let first = route.points.first else { continue }
                cg.beginPath(); cg.move(to: first.cgPoint)
                route.points.dropFirst().forEach { cg.addLine(to: $0.cgPoint) }
                cg.setStrokeColor(UIColor.systemGreen.cgColor); cg.setLineWidth(3); cg.strokePath()
            }
            for device in project.devices {
                let p = device.position.cgPoint
                if device.product.kind.discipline == .video {
                    let half = device.fieldOfViewDegrees / 2 * .pi / 180
                    let heading = device.rotationDegrees * .pi / 180
                    cg.move(to: p); cg.addArc(center: p, radius: device.range, startAngle: heading - half, endAngle: heading + half, clockwise: false); cg.closePath()
                    cg.setFillColor(UIColor.systemBlue.withAlphaComponent(0.12).cgColor); cg.fillPath()
                }
                let rect = CGRect(x: p.x - 12, y: p.y - 12, width: 24, height: 24)
                cg.setFillColor(UIColor.white.cgColor); cg.fillEllipse(in: rect)
                cg.setStrokeColor(device.product.kind.discipline.uiColor.cgColor); cg.setLineWidth(3); cg.strokeEllipse(in: rect)
                let attrs: [NSAttributedString.Key: Any] = [.font: UIFont.systemFont(ofSize: 10, weight: .semibold), .foregroundColor: UIColor.label]
                device.label.draw(at: CGPoint(x: p.x + 16, y: p.y - 6), withAttributes: attrs)
            }
            let title = "\(project.name)  •  \(project.client)"
            title.draw(at: CGPoint(x: 28, y: pageSize.height - 38), withAttributes: [.font: UIFont.boldSystemFont(ofSize: 16), .foregroundColor: UIColor.label])
        }
    }

    private static func draw(_ primitive: VectorPrimitive, in cg: CGContext) {
        cg.setStrokeColor(UIColor.systemGray.cgColor); cg.setFillColor(UIColor.label.cgColor); cg.setLineWidth(1)
        switch primitive {
        case let .line(_, a, b): cg.move(to: a.cgPoint); cg.addLine(to: b.cgPoint); cg.strokePath()
        case let .circle(_, c, r): cg.strokeEllipse(in: CGRect(x: c.x - r, y: c.y - r, width: r * 2, height: r * 2))
        case let .polyline(_, points, closed):
            guard let first = points.first else { return }; cg.move(to: first.cgPoint); points.dropFirst().forEach { cg.addLine(to: $0.cgPoint) }; if closed { cg.closePath() }; cg.strokePath()
        case let .text(_, p, value, height): value.draw(at: p.cgPoint, withAttributes: [.font: UIFont.systemFont(ofSize: max(6, height)), .foregroundColor: UIColor.label])
        }
    }
}

    }

    }

// MARK: - SEPDraw/SEPDrawApp/Services/VisioSVGRenderer.swift
import Foundation

/// Creates a standards-based SVG that Microsoft Visio can open or insert as a
/// vector drawing. This avoids mislabeling XML/SVG data as a native VSDX file.
enum VisioSVGRenderer {
    static func render(project: ProjectDocument, width: Double = 1_584, height: Double = 1_224) -> Data {
        var elements: [String] = []
        elements.append("<rect width=\"100%\" height=\"100%\" fill=\"white\"/>")
        elements.append("<g id=\"Imported-CAD\" fill=\"none\" stroke=\"#7a8696\" stroke-width=\"1\">")
        project.importedVectors.forEach { elements.append(svg($0)) }
        elements.append("</g>")

        elements.append("<g id=\"Cable-Routes\" fill=\"none\" stroke=\"#20945c\" stroke-width=\"3\" stroke-dasharray=\"8 4\">")
        for route in project.cableRoutes where route.points.count > 1 {
            let points = route.points.map { "\(number($0.x)),\(number($0.y))" }.joined(separator: " ")
            elements.append("<polyline id=\"\(route.id.uuidString)\" data-name=\"\(xml(route.name))\" points=\"\(points)\"/>")
        }
        elements.append("</g>")

        for discipline in SecurityDiscipline.allCases {
            elements.append("<g id=\"\(xml(discipline.rawValue))\">")
            for device in project.devices where device.product.kind.discipline == discipline {
                if discipline == .video { elements.append(coverage(device)) }
                elements.append(deviceSymbol(device))
            }
            elements.append("</g>")
        }

        let title = xml(project.name)
        elements.append("<g id=\"Title-Block\"><text x=\"28\" y=\"\(number(height - 28))\" font-family=\"Arial, sans-serif\" font-size=\"16\" font-weight=\"700\" fill=\"#071a33\">\(title)</text></g>")
        let body = elements.joined(separator: "\n")
        let document = """
        <?xml version="1.0" encoding="UTF-8"?>
        <svg xmlns="http://www.w3.org/2000/svg" width="\(number(width))" height="\(number(height))" viewBox="0 0 \(number(width)) \(number(height))">
        <title>\(title)</title>
        <desc>Security drawing exported by SEP Draw for Microsoft Visio</desc>
        \(body)
        </svg>
        """
        return Data(document.utf8)
    }

    private static func svg(_ primitive: VectorPrimitive) -> String {
        switch primitive {
        case let .line(id, a, b):
            return "<line id=\"\(id.uuidString)\" x1=\"\(number(a.x))\" y1=\"\(number(a.y))\" x2=\"\(number(b.x))\" y2=\"\(number(b.y))\"/>"
        case let .circle(id, center, radius):
            return "<circle id=\"\(id.uuidString)\" cx=\"\(number(center.x))\" cy=\"\(number(center.y))\" r=\"\(number(radius))\"/>"
        case let .polyline(id, points, closed):
            let value = points.map { "\(number($0.x)),\(number($0.y))" }.joined(separator: " ")
            return closed ? "<polygon id=\"\(id.uuidString)\" points=\"\(value)\"/>" : "<polyline id=\"\(id.uuidString)\" points=\"\(value)\"/>"
        case let .text(id, origin, value, height):
            return "<text id=\"\(id.uuidString)\" x=\"\(number(origin.x))\" y=\"\(number(origin.y))\" font-family=\"Arial, sans-serif\" font-size=\"\(number(max(6, height)))\" fill=\"#394657\" stroke=\"none\">\(xml(value))</text>"
        }
    }

    private static func coverage(_ device: PlacedDevice) -> String {
        let heading = device.rotationDegrees * .pi / 180
        let half = device.fieldOfViewDegrees * .pi / 360
        let start = WorldPoint(x: device.position.x + cos(heading - half) * device.range, y: device.position.y + sin(heading - half) * device.range)
        let end = WorldPoint(x: device.position.x + cos(heading + half) * device.range, y: device.position.y + sin(heading + half) * device.range)
        let largeArc = device.fieldOfViewDegrees > 180 ? 1 : 0
        return "<path d=\"M \(number(device.position.x)) \(number(device.position.y)) L \(number(start.x)) \(number(start.y)) A \(number(device.range)) \(number(device.range)) 0 \(largeArc) 1 \(number(end.x)) \(number(end.y)) Z\" fill=\"#2f7ed8\" fill-opacity=\"0.12\" stroke=\"#2f7ed8\" stroke-opacity=\"0.45\"/>"
    }

    private static func deviceSymbol(_ device: PlacedDevice) -> String {
        let color: String
        switch device.product.kind.discipline {
        case .video: color = "#2878c8"
        case .access: color = "#d87c18"
        case .intrusion: color = "#c73b3b"
        case .infrastructure: color = "#25875a"
        }
        let x = number(device.position.x), y = number(device.position.y)
        let label = xml(device.label), description = xml(device.product.description)
        return """
        <g id="\(device.id.uuidString)" data-label="\(label)" data-part-number="\(xml(device.product.partNumber))">
          <title>\(label) - \(description)</title>
          <circle cx="\(x)" cy="\(y)" r="17" fill="white" stroke="\(color)" stroke-width="3"/>
          <circle cx="\(x)" cy="\(y)" r="6" fill="\(color)"/>
          <text x="\(number(device.position.x + 22))" y="\(number(device.position.y + 5))" font-family="Arial, sans-serif" font-size="11" font-weight="700" fill="#1c2632">\(label)</text>
        </g>
        """
    }

    private static func number(_ value: Double) -> String { String(format: "%.3f", locale: Locale(identifier: "en_US_POSIX"), value) }
    private static func xml(_ value: String) -> String {
        value.replacingOccurrences(of: "&", with: "&amp;")
            .replacingOccurrences(of: "<", with: "&lt;")
            .replacingOccurrences(of: ">", with: "&gt;")
            .replacingOccurrences(of: "\"", with: "&quot;")
            .replacingOccurrences(of: "'", with: "&apos;")
    }
}

// MARK: - SEPDraw/SEPDrawApp/ViewModels/WorkspaceStore.swift
import Foundation
import Observation
import UniformTypeIdentifiers

@Observable
final class WorkspaceStore {
    var project: ProjectDocument = .sample
    var selectedTool: CanvasTool = .select
    var selectedDeviceID: UUID?
    var selectedProduct: DeviceProduct = ProductCatalog.defaults[0]
    var selectedDiscipline: SecurityDiscipline = .video
    var activeLayerID: UUID?
    var zoom: CGFloat = 1
    var pan: CGSize = .zero
    var inspectorVisible = true
    var libraryVisible = true
    var importPresented = false
    var exportPresented = false
    var alert: WorkspaceAlert?
    var pendingCablePoints: [WorldPoint] = []
    var measurementPoints: [WorldPoint] = []

    private var undoStack: [ProjectDocument] = []
    private var redoStack: [ProjectDocument] = []
    private let storage = ProjectStorage()

    init() { activeLayerID = project.layers.first?.id }

    var selectedDevice: PlacedDevice? {
        guard let selectedDeviceID else { return nil }
        return project.devices.first(where: { $0.id == selectedDeviceID })
    }

    var bom: [BOMLine] { BOMEngine.build(project: project) }
    var materialTotal: Decimal { bom.reduce(0) { $0 + $1.material } }
    var laborHours: Double { bom.reduce(0) { $0 + $1.laborHours } }

    func checkpoint() {
        undoStack.append(project)
        if undoStack.count > 50 { undoStack.removeFirst() }
        redoStack.removeAll()
    }

    func newProject() {
        checkpoint()
        project = .sample
        activeLayerID = project.layers.first?.id
    }

    func addDevice(at point: CGPoint) {
        guard let layerID = activeLayerID ?? project.layers.first?.id else { return }
        checkpoint()
        let prefix: String
        switch selectedProduct.kind.discipline {
        case .video: prefix = "CAM"
        case .access: prefix = "AC"
        case .intrusion: prefix = "IDS"
        case .infrastructure: prefix = "INF"
        }
        let count = project.devices.filter { $0.product.kind.discipline == selectedProduct.kind.discipline }.count + 1
        let device = PlacedDevice(product: selectedProduct, position: WorldPoint(point), label: String(format: "%@-%02d", prefix, count), layerID: layerID)
        project.devices.append(device)
        selectedDeviceID = device.id
        touch()
    }

    func moveDevice(id: UUID, to point: CGPoint, begin: Bool = false) {
        if begin { checkpoint() }
        guard let index = project.devices.firstIndex(where: { $0.id == id }) else { return }
        project.devices[index].position = WorldPoint(point)
        touch()
    }

    func updateSelected(_ transform: (inout PlacedDevice) -> Void) {
        guard let id = selectedDeviceID, let index = project.devices.firstIndex(where: { $0.id == id }) else { return }
        checkpoint()
        transform(&project.devices[index])
        touch()
    }

    func deleteSelection() {
        guard let selectedDeviceID else { return }
        checkpoint()
        project.devices.removeAll { $0.id == selectedDeviceID }
        self.selectedDeviceID = nil
        touch()
    }

    func addCablePoint(_ point: CGPoint) { pendingCablePoints.append(WorldPoint(point)) }

    func finishCable() {
        guard pendingCablePoints.count > 1, let layerID = activeLayerID else { pendingCablePoints.removeAll(); return }
        checkpoint()
        project.cableRoutes.append(.init(name: "Cable \(project.cableRoutes.count + 1)", type: .cat6, points: pendingCablePoints, layerID: layerID))
        pendingCablePoints.removeAll()
        touch()
    }

    func addLayer() {
        checkpoint()
        let layer = DrawingLayer(name: "Layer \(project.layers.count + 1)")
        project.layers.append(layer)
        activeLayerID = layer.id
    }

    func undo() {
        guard let previous = undoStack.popLast() else { return }
        redoStack.append(project); project = previous
    }

    func redo() {
        guard let next = redoStack.popLast() else { return }
        undoStack.append(project); project = next
    }

    func save() {
        do { try storage.save(project); alert = .init(title: "Saved", message: "Project saved on this iPad.") }
        catch { alert = .init(title: "Save Failed", message: error.localizedDescription) }
    }

    func importFile(url: URL) async {
        do {
            let imported = try await FileExchangeService().importFile(url: url, into: project)
            checkpoint(); project = imported; activeLayerID = project.layers.first?.id; touch()
            if ["DWG", "VSD", "VSDX"].contains(project.sourceFileType ?? "") {
                alert = .init(title: "Source Attached", message: "The original \(project.sourceFileType ?? "CAD") file is retained for lossless round-trip export. Configure the licensed interoperability adapter to convert its editable vectors.")
            }
        } catch { alert = .init(title: "Import Failed", message: error.localizedDescription) }
    }

    private func touch() { project.modifiedAt = Date() }
}

    let title: String
    let message: String
}

// MARK: - SEPDraw/SEPDrawApp/Views/DrawingWorkspaceView.swift
import SwiftUI
import PDFKit

struct DrawingWorkspaceView: View {
    @Environment(WorkspaceStore.self) private var workspace
    @State private var movingDeviceID: UUID?
    @State private var pencilVisible = false

    var body: some View {
        @Bindable var workspace = workspace
        GeometryReader { proxy in
            ZStack(alignment: .bottomLeading) {
                Color(uiColor: .secondarySystemBackground)
                ScrollView([.horizontal, .vertical]) {
                    ZStack {
                        drawingSurface(size: CGSize(width: 1_584, height: 1_224))
                        if workspace.selectedTool == .pencil || pencilVisible {
                            PencilOverlay(data: Binding(get: { workspace.project.pencilDrawingData }, set: { workspace.project.pencilDrawingData = $0 }))
                                .allowsHitTesting(workspace.selectedTool == .pencil)
                        }
                    }
                    .frame(width: 1_584, height: 1_224)
                    .scaleEffect(workspace.zoom, anchor: .topLeading)
                    .frame(width: 1_584 * workspace.zoom, height: 1_224 * workspace.zoom, alignment: .topLeading)
                    .background(.white)
                    .shadow(color: .black.opacity(0.16), radius: 12, y: 4)
                    .padding(60)
                }
                StatusHUD(pencilVisible: $pencilVisible)
                    .padding()
            }
            .gesture(MagnifyGesture().onChanged { workspace.zoom = min(4, max(0.25, $0.magnification)) })
        }
    }

    @ViewBuilder private func drawingSurface(size: CGSize) -> some View {
        ZStack {
            if let image = pdfBackground(size: size) { Image(uiImage: image).resizable().scaledToFit().opacity(0.78) }
            Canvas { context, canvasSize in
                drawGrid(context: &context, size: canvasSize)
                workspace.project.importedVectors.forEach { draw($0, context: &context) }
                workspace.project.cableRoutes.filter(layerVisible).forEach { route in
                    var path = Path(); if let first = route.points.first { path.move(to: first.cgPoint); route.points.dropFirst().forEach { path.addLine(to: $0.cgPoint) } }
                    context.stroke(path, with: .color(.green), style: StrokeStyle(lineWidth: 3, dash: [8, 4]))
                }
                if workspace.pendingCablePoints.count > 1 {
                    var path = Path(); path.move(to: workspace.pendingCablePoints[0].cgPoint); workspace.pendingCablePoints.dropFirst().forEach { path.addLine(to: $0.cgPoint) }
                    context.stroke(path, with: .color(.mint), style: StrokeStyle(lineWidth: 3, dash: [5, 5]))
                }
                if workspace.measurementPoints.count == 2 {
                    let a = workspace.measurementPoints[0], b = workspace.measurementPoints[1]
                    var measure = Path(); measure.move(to: a.cgPoint); measure.addLine(to: b.cgPoint)
                    context.stroke(measure, with: .color(.purple), style: StrokeStyle(lineWidth: 2, dash: [4, 3]))
                    let feet = hypot(b.x - a.x, b.y - a.y) * workspace.project.scaleFeetPerPoint
                    context.draw(context.resolve(Text("\(feet, format: .number.precision(.fractionLength(1))) ft").font(.caption.bold()).foregroundStyle(.purple)), at: CGPoint(x: (a.x+b.x)/2, y: (a.y+b.y)/2 - 12))
                }
                workspace.project.devices.filter(layerVisible).forEach { drawDevice($0, context: &context) }
            }
            .contentShape(Rectangle())
            .gesture(canvasGesture)
            VStack { Spacer(); HStack { Spacer(); ScaleLegend().padding(14) } }
        }
        .background(Color.white)
        .clipShape(Rectangle())
    }

    private var canvasGesture: some Gesture {
        DragGesture(minimumDistance: 0)
            .onChanged { value in
                guard workspace.selectedTool == .select else { return }
                let point = CGPoint(x: value.location.x, y: value.location.y)
                if movingDeviceID == nil {
                    movingDeviceID = nearestDevice(to: point)?.id
                    if let id = movingDeviceID { workspace.selectedDeviceID = id; workspace.moveDevice(id: id, to: point, begin: true) }
                }
                if let id = movingDeviceID { workspace.moveDevice(id: id, to: point, begin: false) }
            }
            .onEnded { value in
                let point = CGPoint(x: value.location.x, y: value.location.y)
                switch workspace.selectedTool {
                case .device: workspace.addDevice(at: point)
                case .select:
                    if movingDeviceID == nil { workspace.selectedDeviceID = nearestDevice(to: point)?.id }
                    movingDeviceID = nil
                case .cable:
                    workspace.addCablePoint(point)
                    if abs(value.translation.width) + abs(value.translation.height) > 80 { workspace.finishCable() }
                case .measure:
                    workspace.measurementPoints.append(WorldPoint(point)); if workspace.measurementPoints.count > 2 { workspace.measurementPoints.removeFirst() }
                default: break
                }
            }
    }

    private func nearestDevice(to point: CGPoint) -> PlacedDevice? {
        workspace.project.devices.min { a, b in hypot(a.position.x - point.x, a.position.y - point.y) < hypot(b.position.x - point.x, b.position.y - point.y) }.flatMap { hypot($0.position.x - point.x, $0.position.y - point.y) < 34 ? $0 : nil }
    }

    private func layerVisible(_ device: PlacedDevice) -> Bool { workspace.project.layers.first(where: { $0.id == device.layerID })?.isVisible ?? true }
    private func layerVisible(_ route: CableRoute) -> Bool { workspace.project.layers.first(where: { $0.id == route.layerID })?.isVisible ?? true }

    private func drawGrid(context: inout GraphicsContext, size: CGSize) {
        var minor = Path(); stride(from: 0.0, through: size.width, by: 20).forEach { minor.move(to: .init(x: $0, y: 0)); minor.addLine(to: .init(x: $0, y: size.height)) }; stride(from: 0.0, through: size.height, by: 20).forEach { minor.move(to: .init(x: 0, y: $0)); minor.addLine(to: .init(x: size.width, y: $0)) }
        context.stroke(minor, with: .color(.gray.opacity(0.12)), lineWidth: 0.5)
    }

    private func drawDevice(_ device: PlacedDevice, context: inout GraphicsContext) {
        let point = device.position.cgPoint
        if device.product.kind.discipline == .video {
            let heading = Angle(degrees: device.rotationDegrees).radians, half = Angle(degrees: device.fieldOfViewDegrees / 2).radians
            var coverage = Path(); coverage.move(to: point); coverage.addArc(center: point, radius: device.range, startAngle: .radians(heading - half), endAngle: .radians(heading + half), clockwise: false); coverage.closeSubpath()
            context.fill(coverage, with: .color(.blue.opacity(0.12))); context.stroke(coverage, with: .color(.blue.opacity(0.35)), lineWidth: 1)
        }
        let selected = workspace.selectedDeviceID == device.id
        let bounds = CGRect(x: point.x - 17, y: point.y - 17, width: 34, height: 34)
        context.fill(Path(ellipseIn: bounds), with: .color(.white))
        context.stroke(Path(ellipseIn: bounds), with: .color(device.product.kind.discipline.color), lineWidth: selected ? 5 : 3)
        let icon = context.resolve(Image(systemName: device.product.kind.symbol)); context.draw(icon, at: point)
        let label = context.resolve(Text(device.label).font(.caption2.bold()).foregroundStyle(.primary)); context.draw(label, at: CGPoint(x: point.x, y: point.y + 27))
    }

    private func draw(_ primitive: VectorPrimitive, context: inout GraphicsContext) {
        var path = Path()
        switch primitive {
        case let .line(_, a, b): path.move(to: a.cgPoint); path.addLine(to: b.cgPoint); context.stroke(path, with: .color(.gray), lineWidth: 1)
        case let .circle(_, c, r): context.stroke(Path(ellipseIn: CGRect(x: c.x-r, y: c.y-r, width: r*2, height: r*2)), with: .color(.gray), lineWidth: 1)
        case let .polyline(_, points, closed): guard let first = points.first else { return }; path.move(to: first.cgPoint); points.dropFirst().forEach { path.addLine(to: $0.cgPoint) }; if closed { path.closeSubpath() }; context.stroke(path, with: .color(.gray), lineWidth: 1)
        case let .text(_, p, value, _): context.draw(context.resolve(Text(value).font(.caption).foregroundStyle(.gray)), at: p.cgPoint, anchor: .leading)
        }
    }

    private func pdfBackground(size: CGSize) -> UIImage? {
        guard let data = workspace.project.backgroundPDFData, let page = PDFDocument(data: data)?.page(at: 0) else { return nil }
        return page.thumbnail(of: size, for: .mediaBox)
    }
}

private struct PencilOverlay: UIViewRepresentable {
    @Binding var data: Data?
    func makeCoordinator() -> Coordinator { Coordinator(data: $data) }
    func makeUIView(context: Context) -> PKCanvasView {
        let view = PKCanvasView(); view.backgroundColor = .clear; view.isOpaque = false; view.drawingPolicy = .anyInput; view.delegate = context.coordinator
        if let data, let drawing = try? PKDrawing(data: data) { view.drawing = drawing }
        return view
    }
    func updateUIView(_ view: PKCanvasView, context: Context) {}
    final class Coordinator: NSObject, PKCanvasViewDelegate {
        var data: Binding<Data?>
        init(data: Binding<Data?>) { self.data = data }
        func canvasViewDrawingDidChange(_ canvasView: PKCanvasView) { data.wrappedValue = canvasView.drawing.dataRepresentation() }
    }
}

// MARK: - SEPDraw/SEPDrawApp/Views/RootView.swift
import SwiftUI
import UniformTypeIdentifiers

struct RootView: View {
    @Environment(WorkspaceStore.self) private var workspace
    @State private var exportDocument: BinaryExportDocument?
    @State private var exportType: UTType = .data
    @State private var exportName = "SEP-Draw-Export"
    @State private var exportActive = false
    @State private var showExportMenu = false
    @State private var showBOM = false

    var body: some View {
        @Bindable var workspace = workspace
        NavigationSplitView {
            DeviceLibraryView()
                .navigationTitle("SEP Draw")
        } detail: {
            VStack(spacing: 0) {
                ProjectHeaderView()
                Divider()
                HStack(spacing: 0) {
                    DrawingWorkspaceView()
                    if workspace.inspectorVisible {
                        Divider()
                        InspectorView()
                            .frame(width: 310)
                    }
                }
            }
            .toolbar { toolbarContent }
        }
        .navigationSplitViewStyle(.balanced)
        .fileImporter(isPresented: $workspace.importPresented, allowedContentTypes: [.sepDrawProject, .pdf, .dxf, .dwg, .vsdx, .data], allowsMultipleSelection: false) { result in
            guard case .success(let urls) = result, let url = urls.first else {
                if case .failure(let error) = result { workspace.alert = .init(title: "Import Failed", message: error.localizedDescription) }
                return
            }
            Task { await workspace.importFile(url: url) }
        }
        .fileExporter(isPresented: $exportActive, document: exportDocument, contentType: exportType, defaultFilename: exportName) { result in
            if case .failure(let error) = result { workspace.alert = .init(title: "Export Failed", message: error.localizedDescription) }
        }
        .sheet(isPresented: $showExportMenu) { ExportCenterView(export: prepareExport) }
        .sheet(isPresented: $showBOM) { NavigationStack { BOMView().navigationTitle("Bill of Materials") } }
        .alert(item: $workspace.alert) { item in Alert(title: Text(item.title), message: Text(item.message), dismissButton: .default(Text("OK"))) }
    }

    @ToolbarContentBuilder private var toolbarContent: some ToolbarContent {
        ToolbarItemGroup(placement: .topBarLeading) {
            Button { workspace.undo() } label: { Label("Undo", systemImage: "arrow.uturn.backward") }.keyboardShortcut("z", modifiers: .command)
            Button { workspace.redo() } label: { Label("Redo", systemImage: "arrow.uturn.forward") }.keyboardShortcut("z", modifiers: [.command, .shift])
        }
        ToolbarItemGroup(placement: .principal) {
            ToolStripView()
        }
        ToolbarItemGroup(placement: .topBarTrailing) {
            Button { workspace.importPresented = true } label: { Label("Import", systemImage: "square.and.arrow.down") }.keyboardShortcut("o", modifiers: .command)
            Button { showExportMenu = true } label: { Label("Export", systemImage: "square.and.arrow.up") }
            Button { showBOM = true } label: { Label("BOM", systemImage: "list.bullet.rectangle") }
            Button { workspace.inspectorVisible.toggle() } label: { Label("Inspector", systemImage: "sidebar.right") }
        }
    }

    private func prepareExport(_ kind: ExportKind) {
        switch kind {
        case .visio:
            exportType = .visioSVG; exportName = "\(workspace.project.name)-Visio"; exportDocument = .init(data: VisioSVGRenderer.render(project: workspace.project), type: exportType)
        case .pdf:
            exportType = .pdf; exportName = workspace.project.name; exportDocument = .init(data: PDFDrawingRenderer.render(project: workspace.project), type: exportType)
        case .autocad:
            exportType = .dxf; exportName = "\(workspace.project.name)-AutoCAD"; exportDocument = .init(data: DXFCodec().encode(project: workspace.project), type: exportType)
        }
        showExportMenu = false; exportActive = true
    }
}
